// ── Revert task-runner timeout from 600s back to 60s ─────────────────────────
// kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=60
// Verify: kubectl exec -it deployment/n8n -n preme-n8n -c task-runner -- env | grep TASK_TIMEOUT

// n8n Code Node — Full login + article list + content debug
// 50s overall timeout fires before runner's 60s limit (gives 10s cleanup)
// currentStep updated at each point — shows exactly where it got stuck

const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const EMAIL = 'YOUR_EMAIL';
const PASSWORD = 'YOUR_PASSWORD';
const LIST_URL = 'https://background.tagesspiegel.de/digitalisierung-und-ki';

const wait = ms => new Promise(r => setTimeout(r, ms));

let currentStep = 'init';
let browser = null;
let page = null;

// Overall 50s timeout — gives 10s for cleanup before runner kills at 60s
const overallTimeout = wait(50000).then(() => {
  throw new Error(`OVERALL TIMEOUT (50s) — stuck at: [${currentStep}]`);
});

const main = async () => {

  currentStep = 'launch browser';
  browser = await puppeteer.launch({
    executablePath: '/usr/lib/chromium/chromium',
    headless: true,
    ignoreHTTPSErrors: true,
    args: [
      '--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage',
      '--ignore-certificate-errors',
      '--proxy-server=http://proxy.idst.ibaintern.de:8080'
    ]
  });

  currentStep = 'new page';
  page = await browser.newPage();
  page.setDefaultTimeout(0);
  page.setDefaultNavigationTimeout(0);
  await page.setExtraHTTPHeaders({ 'Accept-Language': 'de-DE,de;q=0.9' });

  // ── Login ──────────────────────────────────────────────────────────────────
  currentStep = 'goto login page';
  await page.goto('https://background.tagesspiegel.de/login', { waitUntil: 'domcontentloaded', timeout: 0 });
  currentStep = 'login page loaded url=' + page.url();

  currentStep = 'fill email';
  await page.evaluate(`(function() {
    var input = document.querySelector('input[type=email]') || document.querySelector('input[name=email]');
    if (!input) throw new Error('email input not found');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${EMAIL.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'fill password';
  await page.evaluate(`(function() {
    var input = document.querySelector('input[type=password]');
    if (!input) throw new Error('password input not found');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${PASSWORD.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'click submit';
  await page.evaluate(`(function() {
    var btn = document.querySelector('button[type=submit]');
    if (!btn) throw new Error('submit button not found');
    btn.click();
  })()`);

  currentStep = 'waiting 8s after submit';
  await wait(8000);
  currentStep = 'login done url=' + page.url();

  // ── Article List ───────────────────────────────────────────────────────────
  currentStep = 'goto list page';
  await page.goto(LIST_URL, { waitUntil: 'domcontentloaded', timeout: 0 });
  currentStep = 'list page loaded url=' + page.url();

  currentStep = 'count articles';
  const articleCount = await page.evaluate(`(function() {
    return document.querySelectorAll('div.ts-article-tile').length;
  })()`);
  currentStep = 'articles found=' + articleCount;

  const firstUrl = await page.evaluate(`(function() {
    var a = document.querySelector('div.ts-article-tile a.stretched-link');
    return a ? a.href : 'not found';
  })()`);
  currentStep = 'first article url=' + firstUrl;

  if (firstUrl === 'not found') {
    return `LOGIN OK, LIST OK, but no article tiles found — url=${page.url()}`;
  }

  // ── First Article Content ──────────────────────────────────────────────────
  currentStep = 'goto first article';
  await page.goto(firstUrl, { waitUntil: 'domcontentloaded', timeout: 0 });
  currentStep = 'article page loaded url=' + page.url();

  const articleTitle = await page.evaluate(`(function() {
    return document.querySelector('h1')?.innerText?.trim() || 'not found';
  })()`);

  const articleLead = await page.evaluate(`(function() {
    return document.querySelector('p')?.innerText?.trim()?.substring(0, 200) || 'not found';
  })()`);

  return `SUCCESS\nlogin=OK\narticles=${articleCount}\nfirst=${firstUrl}\ntitle=${articleTitle}\nlead=${articleLead}`;
};

try {
  const result = await Promise.race([main(), overallTimeout]);
  throw new Error(result);
} catch(e) {
  if (browser) await Promise.race([browser.close(), wait(3000)]).catch(() => {});
  throw new Error(e.message);
}
