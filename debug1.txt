// ── Set task-runner timeout to 120s for this debug ───────────────────────────
// kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=120
// To revert: kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=60

const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const EMAIL = 'YOUR_EMAIL';
const PASSWORD = 'YOUR_PASSWORD';
const LIST_URL = 'https://background.tagesspiegel.de/digitalisierung-und-ki';

const wait = ms => new Promise(r => setTimeout(r, ms));

let currentStep = 'init';
let browser = null;
let page = null;
let httpStatus = 'NO_RESPONSE';
let firstByteMs = null;

const overallTimeout = wait(110000).then(() => {
  throw new Error(`OVERALL TIMEOUT (110s) — stuck at: [${currentStep}]`);
});

const main = async () => {
  currentStep = 'launch browser';
  browser = await puppeteer.launch({
    executablePath: '/usr/lib/chromium/chromium',
    headless: true,
    ignoreHTTPSErrors: true,
    args: [
      '--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage',
      '--ignore-certificate-errors',
      '--proxy-server=http://proxy.idst.ibaintern.de:8080'
    ]
  });

  page = await browser.newPage();
  page.setDefaultTimeout(0);
  page.setDefaultNavigationTimeout(0);
  await page.setExtraHTTPHeaders({ 'Accept-Language': 'de-DE,de;q=0.9' });

  const navStart = Date.now();
  page.on('response', res => {
    if (res.url().includes('tagesspiegel.de') && firstByteMs === null) {
      firstByteMs = Date.now() - navStart;
      httpStatus = res.status();
    }
  });

  // Single interceptor with a flag — swapping listeners mid-navigation causes races.
  let blockScripts = true;
  await page.setRequestInterception(true);
  page.on('request', req => {
    const t = req.resourceType();
    if (blockScripts) {
      // Login page: block scripts so DOMContentLoaded fires (scripts hang via SSL inspection)
      if (['image', 'stylesheet', 'font', 'media', 'script', 'other'].includes(t)) {
        req.abort();
      } else {
        req.continue();
      }
    } else {
      // List page: allow scripts for article tile rendering
      if (['image', 'font', 'media'].includes(t)) {
        req.abort();
      } else {
        req.continue();
      }
    }
  });

  // page.goto() can throw "Navigating frame was detached" when a redirect chain
  // crosses origins and Chromium replaces the renderer process. The navigation
  // still succeeds — the new frame just needs a moment to settle.
  const safegoto = async (url, maxMs, label) => {
    try {
      await Promise.race([
        page.goto(url, { waitUntil: 'domcontentloaded', timeout: 0 }),
        wait(maxMs).then(() => { throw new Error(`[${label}] DOMContentLoaded timed out ${maxMs}ms — url=${page.url()}`); })
      ]);
    } catch(e) {
      if ((e.message || '').toLowerCase().includes('detach')) {
        // Frame replaced — poll evaluate until the new frame is ready
        currentStep = `[${label}] frame detached, waiting for new frame`;
        const since = Date.now();
        while (Date.now() - since < 15000) {
          try { await page.evaluate('1'); break; } catch(_) { await wait(500); }
        }
      } else {
        throw e;
      }
    }
  };

  // ── Login ──────────────────────────────────────────────────────────────────
  currentStep = 'goto login page';
  await safegoto('https://background.tagesspiegel.de/login', 30000, 'login');

  currentStep = `login loaded — httpStatus=${httpStatus} firstByteMs=${firstByteMs} url=${page.url()}`;

  const hasEmail = await page.evaluate(`!!(document.querySelector('input[type=email]') || document.querySelector('input[name="_username"]') || document.querySelector('input[name="email"]'))`);

  if (!hasEmail) {
    const title  = await page.evaluate(`document.title`).catch(() => 'eval-fail');
    const body   = await page.evaluate(`document.body ? document.body.innerText.slice(0,400).replace(/\\s+/g,' ') : 'no body'`).catch(() => 'eval-fail');
    const frames = page.frames().map(f => f.url()).join(' | ');
    throw new Error(`email input not in DOM.\ntitle="${title}" frames=[${frames}]\nbody="${body}"`);
  }

  currentStep = 'fill email';
  await page.evaluate(`(function() {
    var input = document.querySelector('input[type=email]') || document.querySelector('input[name="_username"]') || document.querySelector('input[name="email"]');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${EMAIL.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'fill password';
  await page.evaluate(`(function() {
    var input = document.querySelector('input[type=password]') || document.querySelector('input[name="_password"]');
    if (!input) throw new Error('password input not found');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${PASSWORD.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'click submit';
  await page.evaluate(`document.querySelector('button[type=submit]').click()`);

  // Poll URL — no waitForNavigation to avoid another detach error
  currentStep = 'waiting for post-login redirect';
  const redirectStart = Date.now();
  while (page.url().includes('/login') && Date.now() - redirectStart < 30000) {
    await wait(500);
  }
  if (page.url().includes('/login')) throw new Error(`still on login after 30s — login failed. url=${page.url()}`);
  currentStep = 'login done url=' + page.url();

  blockScripts = false;

  // ── Article List ───────────────────────────────────────────────────────────
  currentStep = 'goto list page';
  await safegoto(LIST_URL, 30000, 'list');

  currentStep = 'waiting for article tiles';
  let articleCount = 0;
  const listPollStart = Date.now();
  while (Date.now() - listPollStart < 20000) {
    const n = await page.evaluate(`document.querySelectorAll('div.ts-article-tile').length`).catch(() => 0);
    if (n > 0) { articleCount = n; break; }
    await wait(1000);
  }

  const firstUrl = await page.evaluate(`(function() {
    var a = document.querySelector('div.ts-article-tile a.stretched-link');
    return a ? a.href : 'not found';
  })()`).catch(() => 'eval-fail');

  return `SUCCESS\nhttpStatus=${httpStatus} firstByteMs=${firstByteMs}\nlogin=OK\npost-login url=${page.url()}\narticles=${articleCount}\nfirst=${firstUrl}`;
};

try {
  const result = await Promise.race([main(), overallTimeout]);
  throw new Error(result);
} catch(e) {
  if (browser) await Promise.race([browser.close(), wait(2000)]).catch(() => {});
  throw new Error(e.message);
}
