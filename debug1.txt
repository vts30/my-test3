// Stage 1 — Test login only (paste as Code node in n8n UI)
// Replace YOUR_EMAIL and YOUR_PASSWORD with real credentials

const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const EMAIL = 'YOUR_EMAIL';
const PASSWORD = 'YOUR_PASSWORD';

const wait = ms => new Promise(r => setTimeout(r, ms));
const raceTimeout = (promise, ms, msg) => Promise.race([
  promise,
  wait(ms).then(() => { throw new Error(msg); })
]);

const browser = await puppeteer.launch({
  executablePath: '/usr/lib/chromium/chromium',
  args: ['--no-sandbox','--disable-setuid-sandbox','--disable-dev-shm-usage',
         '--ignore-certificate-errors','--proxy-server=http://proxy.idst.ibaintern.de:8080'],
  headless: true, ignoreHTTPSErrors: true
});

let result = 'not started';
try {
  const page = await browser.newPage();
  page.setDefaultNavigationTimeout(0);
  page.setDefaultTimeout(0); // disable all puppeteer timeouts to avoid runner crash

  await page.setExtraHTTPHeaders({ 'Accept-Language': 'de-DE,de;q=0.9' });

  result = 'before goto login';
  await raceTimeout(
    page.goto('https://background.tagesspiegel.de/login', { waitUntil: 'domcontentloaded' }),
    20000, 'HUNG: page.goto login'
  );

  result = 'login page loaded url=' + page.url();
  await wait(3000);

  result = 'filling email';
  await raceTimeout(page.evaluate(`(function() {
    var input = document.querySelector('input[type=email]');
    if (!input) throw new Error('no email input');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${EMAIL.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`), 10000, 'HUNG: evaluate email');

  result = 'filling password';
  await raceTimeout(page.evaluate(`(function() {
    var input = document.querySelector('input[type=password]');
    if (!input) throw new Error('no password input');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${PASSWORD.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`), 10000, 'HUNG: evaluate password');

  result = 'clicking submit';
  await raceTimeout(page.click('button[type=submit]'), 5000, 'HUNG: click submit');

  await wait(8000);
  result = 'URL after login: ' + page.url();
} catch(e) {
  result = 'FAIL at [' + result + ']: ' + e.message;
} finally {
  await Promise.race([browser.close(), wait(5000)]);
}

throw new Error(result);
