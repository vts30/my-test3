// ── Pre-run setup (run in terminal before executing this node) ────────────────
// Set timeout to 120s:
//    kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=120
//    kubectl rollout restart deployment/n8n -n preme-n8n
// Revert after debug:
//    kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=60

const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const EMAIL = 'YOUR_EMAIL';
const PASSWORD = 'YOUR_PASSWORD';
const LIST_URL = 'https://background.tagesspiegel.de/digitalisierung-und-ki';

const wait = ms => new Promise(r => setTimeout(r, ms));

let currentStep = 'init';
let browser = null;
let page = null;
let httpStatus = 'NO_RESPONSE';
let firstByteMs = null;

const overallTimeout = wait(110000).then(() => {
  throw new Error(`OVERALL TIMEOUT (110s) — stuck at: [${currentStep}]`);
});

const main = async () => {
  currentStep = 'launch browser';
  browser = await puppeteer.launch({
    executablePath: '/usr/lib/chromium/chromium',
    headless: 'shell',
    ignoreHTTPSErrors: true,
    env: {
      HOME: '/tmp',
      XDG_CONFIG_HOME: '/tmp/.config',
      XDG_CACHE_HOME: '/tmp/.cache',
      HTTP_PROXY: 'http://proxy.idst.ibaintern.de:8080',
      HTTPS_PROXY: 'http://proxy.idst.ibaintern.de:8080',
    },
    args: [
      '--no-sandbox',
      '--disable-setuid-sandbox',
      '--disable-gpu',
      '--ignore-certificate-errors',
      '--proxy-server=http://proxy.idst.ibaintern.de:8080',
      '--user-data-dir=/tmp/chrome-user-data',
    ]
  });

  page = await browser.newPage();
  page.setDefaultTimeout(0);
  page.setDefaultNavigationTimeout(0);
  await page.setExtraHTTPHeaders({ 'Accept-Language': 'de-DE,de;q=0.9' });

  const navStart = Date.now();
  page.on('response', res => {
    if (res.url().includes('tagesspiegel.de') && firstByteMs === null) {
      firstByteMs = Date.now() - navStart;
      httpStatus = res.status();
    }
  });

  // Block scripts on login page so DOMContentLoaded fires fast
  // (scripts hang via proxy SSL inspection, causing frame.evaluate to block forever).
  // blockScripts flipped to false before list page so article tiles can render.
  let blockScripts = true;
  await page.setRequestInterception(true);
  page.on('request', req => {
    const t = req.resourceType();
    if (blockScripts) {
      if (['image', 'stylesheet', 'font', 'media', 'script', 'other'].includes(t)) {
        req.abort().catch(() => {});
      } else {
        req.continue().catch(() => {});
      }
    } else {
      if (['image', 'font', 'media'].includes(t)) {
        req.abort().catch(() => {});
      } else {
        req.continue().catch(() => {});
      }
    }
  });

  // Navigate via window.location.href instead of page.goto() to avoid puppeteer's
  // LifecycleWatcher throwing "Navigating frame was detached" / "Attempted to use
  // detached Frame" when redirect chains cross origins.
  const navigate = async (url) => {
    try {
      await page.evaluate(`window.location.href = '${url.replace(/'/g, "\\'")}'`);
    } catch(e) {}
  };

  // Poll expr every 500ms until truthy or timeout. Silently catches all errors
  // (frame transitioning, context destroyed) and retries.
  const poll = async (expr, maxMs) => {
    const deadline = Date.now() + maxMs;
    while (Date.now() < deadline) {
      try {
        const v = await page.evaluate(expr);
        if (v) return v;
      } catch(e) {}
      await wait(500);
    }
    return null;
  };

  // ── Login ──────────────────────────────────────────────────────────────────
  currentStep = 'navigate to login page';
  await navigate('https://background.tagesspiegel.de/login');

  currentStep = 'waiting for email input';
  const hasEmail = await poll(
    `!!(document.querySelector('input[type=email]') || document.querySelector('input[name="_username"]') || document.querySelector('input[name="email"]'))`,
    45000
  );

  if (!hasEmail) {
    let title = 'eval-fail', body = 'eval-fail', url = 'unknown';
    try { title = await page.evaluate(`document.title`); } catch(e) {}
    try { body = await page.evaluate(`document.body ? document.body.innerText.slice(0,300).replace(/\\s+/g,' ') : 'no body'`); } catch(e) {}
    try { url = await page.evaluate(`window.location.href`); } catch(e) {}
    throw new Error(`email input not found after 45s.\nhttpStatus=${httpStatus} firstByteMs=${firstByteMs}\ntitle="${title}" url=${url}\nbody="${body}"`);
  }

  currentStep = 'fill email';
  await page.evaluate(`(function() {
    var input = document.querySelector('input[type=email]') || document.querySelector('input[name="_username"]') || document.querySelector('input[name="email"]');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${EMAIL.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'fill password';
  await page.evaluate(`(function() {
    var input = document.querySelector('input[type=password]') || document.querySelector('input[name="_password"]');
    if (!input) throw new Error('password input not found');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${PASSWORD.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'click submit';
  await page.evaluate(`document.querySelector('button[type=submit]').click()`);

  currentStep = 'waiting for post-login redirect';
  const redirectStart = Date.now();
  while (Date.now() - redirectStart < 30000) {
    try {
      const u = await page.evaluate(`window.location.href`);
      if (u && !u.includes('/login')) break;
    } catch(e) {}
    await wait(500);
  }
  const postLoginUrl = await page.evaluate(`window.location.href`).catch(() => 'eval-fail');
  if (postLoginUrl.includes('/login')) throw new Error(`still on login after 30s — login failed. url=${postLoginUrl}`);
  currentStep = 'login done url=' + postLoginUrl;

  // ── Article List ───────────────────────────────────────────────────────────
  blockScripts = false;
  currentStep = 'navigate to list page';
  await navigate(LIST_URL);

  currentStep = 'waiting for article tiles';
  const articleCount = await poll(`(function() {
    var n = document.querySelectorAll('div.ts-article-tile').length;
    return n > 0 ? n : null;
  })()`, 25000) || 0;

  const firstUrl = await page.evaluate(`(function() {
    var a = document.querySelector('div.ts-article-tile a.stretched-link');
    return a ? a.href : 'not found';
  })()`).catch(() => 'eval-fail');

  return `SUCCESS\nhttpStatus=${httpStatus} firstByteMs=${firstByteMs}\nlogin=OK\npost-login url=${postLoginUrl}\narticles=${articleCount}\nfirst=${firstUrl}`;
};

try {
  const result = await Promise.race([main(), overallTimeout]);
  throw new Error(result);
} catch(e) {
  if (browser) await Promise.race([browser.close(), wait(2000)]).catch(() => {});
  throw new Error(e.message);
}
