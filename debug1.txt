// ── Minimal crash diagnostic — just launch + close Chromium ──────────────────
// If this crashes: Chromium launch itself fails in task-runner container
// If this succeeds: crash is during navigation (we add steps back after)

const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const wait = ms => new Promise(r => setTimeout(r, ms));
let browser = null;

const main = async () => {
  browser = await puppeteer.launch({
    executablePath: '/usr/lib/chromium/chromium',
    headless: true,
    ignoreHTTPSErrors: true,
    args: [
      '--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage',
      '--ignore-certificate-errors', '--disable-gpu', '--no-zygote',
      '--disable-extensions', '--disable-background-networking'
    ]
  });

  const page = await browser.newPage();
  await wait(3000);
  return 'STEP1 OK: browser launched, page created, waited 3s';
};

try {
  const result = await main();
  throw new Error(result);
} catch(e) {
  if (browser) await Promise.race([browser.close(), wait(2000)]).catch(() => {});
  throw new Error(e.message);
}
