// ── Set task-runner timeout to 120s for this debug ───────────────────────────
// kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=120
// Verify: kubectl exec -it deployment/n8n -n preme-n8n -c task-runner -- env | grep TASK_TIMEOUT
// To revert to 60s after debug: kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=60

// n8n Code Node — Full login + article list + content debug
// Tracks executionContextId explicitly + auto-retries cdpEval

const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const EMAIL = 'YOUR_EMAIL';
const PASSWORD = 'YOUR_PASSWORD';
const LIST_URL = 'https://background.tagesspiegel.de/digitalisierung-und-ki';

const wait = ms => new Promise(r => setTimeout(r, ms));

let currentStep = 'init';
let browser = null;
let page = null;

// Overall 110s timeout — gives 10s cleanup before runner kills at 120s
// Runner must be set to 120s: kubectl set env deployment/n8n -n preme-n8n -c task-runner N8N_RUNNERS_TASK_TIMEOUT=120
const overallTimeout = wait(110000).then(() => {
  throw new Error(`OVERALL TIMEOUT (110s) — stuck at: [${currentStep}]`);
});

const main = async () => {

  currentStep = 'launch browser';
  browser = await puppeteer.launch({
    executablePath: '/usr/lib/chromium/chromium',
    headless: true,
    ignoreHTTPSErrors: true,
    args: [
      '--no-sandbox', '--disable-setuid-sandbox', '--disable-dev-shm-usage',
      '--ignore-certificate-errors',
      '--proxy-server=http://proxy.idst.ibaintern.de:8080'
    ]
  });

  currentStep = 'new page';
  page = await browser.newPage();
  page.setDefaultTimeout(0);
  page.setDefaultNavigationTimeout(0);
  await page.setExtraHTTPHeaders({ 'Accept-Language': 'de-DE,de;q=0.9' });
  const cdp = await page.createCDPSession();

  // Track the latest valid default execution context ID
  let contextId = null;
  let contextReady = false;
  await cdp.send('Runtime.enable');
  cdp.on('Runtime.executionContextCreated', (evt) => {
    // Only track default top-level contexts, not iframes
    if (evt.context.auxData && evt.context.auxData.isDefault) {
      contextId = evt.context.id;
      contextReady = true;
    }
  });
  cdp.on('Runtime.executionContextDestroyed', (evt) => {
    if (evt.executionContextId === contextId) {
      contextId = null;
      contextReady = false;
    }
  });

  // CDP evaluate with explicit contextId — single attempt, 3s timeout
  // Callers that need retry use their own loop (e.g. email input polling)
  const cdpEval = async (expr) => {
    const params = { expression: expr, returnByValue: true };
    if (contextId) params.contextId = contextId;
    const res = await Promise.race([
      cdp.send('Runtime.evaluate', params).catch(e => ({ _error: e.message })),
      wait(3000).then(() => ({ _timeout: true }))
    ]);
    if (res._timeout) throw new Error('cdpEval timeout (3s)');
    if (res._error) throw new Error('cdpEval error: ' + res._error);
    if (res.exceptionDetails) {
      throw new Error(res.exceptionDetails.exception?.description || res.exceptionDetails.text || 'eval exception');
    }
    return res.result ? res.result.value : undefined;
  };

  // Navigate and wait until default JS context is ready
  const gotoAndWaitContext = async (url, maxMs, label) => {
    contextId = null;
    contextReady = false;
    page.goto(url, { waitUntil: 'domcontentloaded', timeout: 0 }).catch(() => {});
    const start = Date.now();
    currentStep = label + ' [waiting for context]';
    while (!contextReady && Date.now() - start < maxMs) {
      await wait(300);
    }
    const elapsed = Date.now() - start;
    if (!contextReady) throw new Error(`${label}: no JS context after ${elapsed}ms — proxy not delivering content`);
    currentStep = label + ` [context in ${elapsed}ms] url=` + page.url();
  };

  // ── Login ──────────────────────────────────────────────────────────────────
  await gotoAndWaitContext('https://background.tagesspiegel.de/login', 30000, 'login page');

  // Poll until email input appears — skip if no context yet, check contextReady first
  currentStep = 'waiting for email input';
  let emailReady = false;
  const emailPollStart = Date.now();
  while (!emailReady && Date.now() - emailPollStart < 15000) {
    if (!contextReady || !contextId) { await wait(300); continue; }
    try {
      const found = await cdpEval(`!!(document.querySelector('input[type=email]') || document.querySelector('input[name=email]'))`);
      if (found) { emailReady = true; break; }
    } catch(e) { /* context changed, retry */ }
    await wait(500);
  }
  if (!emailReady) {
    let diagInfo = 'n/a';
    try {
      diagInfo = await cdpEval(`(function() {
        return JSON.stringify({
          url: location.href,
          title: document.title,
          inputs: [...document.querySelectorAll('input')].map(i => i.type + '/' + i.name + '/' + i.className).join(', '),
          bodyText: document.body ? document.body.innerText.slice(0, 150) : 'no body'
        });
      })()`);
    } catch(e) { diagInfo = 'eval failed: ' + e.message; }
    throw new Error(`email input not found after 15s — ${diagInfo}`);
  }

  currentStep = 'fill email';
  await cdpEval(`(function() {
    var input = document.querySelector('input[type=email]') || document.querySelector('input[name=email]');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${EMAIL.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'fill password';
  await cdpEval(`(function() {
    var input = document.querySelector('input[type=password]');
    if (!input) throw new Error('password input not found');
    var setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
    setter.call(input, '${PASSWORD.replace(/'/g, "\\'")}');
    input.dispatchEvent(new Event('input', { bubbles: true }));
    input.dispatchEvent(new Event('change', { bubbles: true }));
  })()`);

  currentStep = 'click submit';
  contextReady = false;
  await cdpEval(`(function() {
    var btn = document.querySelector('button[type=submit]');
    if (!btn) throw new Error('submit button not found');
    btn.click();
  })()`);

  // Wait for post-submit redirect — just listen for new context, no extra goto
  currentStep = 'post-login redirect [waiting for context]';
  const redirectStart = Date.now();
  while (!contextReady && Date.now() - redirectStart < 25000) {
    await wait(300);
  }
  if (!contextReady) throw new Error(`post-login redirect: no JS context after ${Date.now() - redirectStart}ms`);
  currentStep = 'login done url=' + page.url();

  // ── Article List ───────────────────────────────────────────────────────────
  await gotoAndWaitContext(LIST_URL, 30000, 'list page');

  currentStep = 'count articles';
  const articleCount = await cdpEval(`document.querySelectorAll('div.ts-article-tile').length`);
  currentStep = 'articles found=' + articleCount;

  const firstUrl = await cdpEval(`(function() {
    var a = document.querySelector('div.ts-article-tile a.stretched-link');
    return a ? a.href : 'not found';
  })()`);

  return `SUCCESS\nlogin=OK\npost-login url=${page.url()}\narticles=${articleCount}\nfirst=${firstUrl}`;
};

try {
  const result = await Promise.race([main(), overallTimeout]);
  throw new Error(result);
} catch(e) {
  if (browser) await Promise.race([browser.close(), wait(2000)]).catch(() => {});
  throw new Error(e.message);
}
