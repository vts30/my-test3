const puppeteer = require('puppeteer-extra');
const Stealth = require('puppeteer-extra-plugin-stealth');
puppeteer.use(Stealth());

const EMAIL = 'YOUR_EMAIL';
const PASSWORD = 'YOUR_PASSWORD';

const wait = ms => new Promise(r => setTimeout(r, ms));

const trace = [];

function log(msg) {
    trace.push(`[${new Date().toISOString()}] ${msg}`);
}

async function step(name, fn) {
    log(`START: ${name}`);

    try {
        const result = await Promise.race([
            fn(),
            wait(30000).then(() => {
                throw new Error(`Timeout during "${name}"`);
            })
        ]);

        log(`DONE: ${name}`);
        return result;

    } catch (err) {
        log(`FAILED: ${name}`);
        log(err.stack || err.message);
        throw err;
    }
}

let browser;
let page;

try {

    log("Launching browser");

    browser = await puppeteer.launch({
        executablePath: '/usr/lib/chromium/chromium',
        headless: true,
        ignoreHTTPSErrors: true,
        args: [
            '--no-sandbox',
            '--disable-setuid-sandbox',
            '--disable-dev-shm-usage',
            '--ignore-certificate-errors',
            '--proxy-server=http://proxy.idst.ibaintern.de:8080'
        ]
    });

    log(await browser.version());

    page = await browser.newPage();

    page.setDefaultNavigationTimeout(30000);
    page.setDefaultTimeout(30000);

    await page.setExtraHTTPHeaders({
        'Accept-Language': 'de-DE,de;q=0.9'
    });

    //
    // Browser event logging
    //
    page.on('console', msg => {
        log(`[console:${msg.type()}] ${msg.text()}`);
    });

    page.on('pageerror', err => {
        log(`[pageerror] ${err.message}`);
    });

    page.on('requestfailed', req => {
        const f = req.failure();
        log(`[requestfailed] ${req.url()} :: ${f ? f.errorText : ''}`);
    });

    page.on('response', res => {
        if (res.status() >= 400) {
            log(`[HTTP ${res.status()}] ${res.url()}`);
        }
    });

    async function dump(label) {

        log(`===== ${label} =====`);

        log("URL: " + page.url());

        try {
            log("Title: " + await page.title());
        } catch(e){}

        try {
            const text = await page.evaluate(() => {
                return document.body.innerText.substring(0,3000);
            });

            log("TEXT:\n" + text);

        } catch(e) {
            log("Cannot read page text");
        }

    }

    //
    // Test connectivity
    //
    await step("Open example.com", async () => {

        await page.goto(
            "https://example.com",
            {waitUntil:"domcontentloaded"}
        );

    });

    await dump("example.com");

    //
    // Open login page
    //
    await step("Open Tagesspiegel Login", async () => {

        await page.goto(
            "https://background.tagesspiegel.de/login",
            {
                waitUntil:"domcontentloaded"
            }
        );

    });

    await dump("Login page");

    //
    // Wait for email field
    //
    await step("Wait email input", async () => {

        await page.waitForSelector(
            'input[type=email]',
            {visible:true}
        );

    });

    //
    // Type email
    //
    await step("Type email", async () => {

        await page.type(
            'input[type=email]',
            EMAIL,
            {delay:30}
        );

    });

    //
    // Wait password
    //
    await step("Wait password input", async () => {

        await page.waitForSelector(
            'input[type=password]',
            {visible:true}
        );

    });

    //
    // Type password
    //
    await step("Type password", async () => {

        await page.type(
            'input[type=password]',
            PASSWORD,
            {delay:30}
        );

    });

    //
    // Click submit
    //
    await step("Click submit", async () => {

        await page.click(
            'button[type=submit]'
        );

    });

    //
    // Wait after login
    //
    await wait(10000);

    await dump("After Login");

    throw new Error(
        "SUCCESS\n\n" +
        trace.join("\n")
    );

}
catch(err){

    if(page){

        try{

            log("Capturing screenshot...");

            const img = await page.screenshot({
                encoding:'base64',
                fullPage:true
            });

            log("SCREENSHOT(base64 first 1000 chars):");
            log(img.substring(0,1000));

        }catch(e){
            log("Screenshot failed");
        }

        try{

            const html = await page.content();

            log("HTML(first 5000 chars):");
            log(html.substring(0,5000));

        }catch(e){
            log("HTML capture failed");
        }

    }

    throw new Error(
        err.message +
        "\n\n================ TRACE ================\n\n" +
        trace.join("\n")
    );

}
finally{

    if(browser){

        try{
            await browser.close();
        }catch(e){}
    }

}
